Código Google Apps Script

Para sincronizar automáticamente todas las inscripciones realizadas por los colegios con tu planilla de Google Sheets, abre tu planilla, dirígete a Extensiones > Apps Script, pega el código siguiente e impleméntalo como Aplicación Web. 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  sheet.appendRow([
    data.codigoTracking,
    data.sede,
    data.rbd,
    data.nombreColegio,
    data.comuna,
    data.cursosResumenTexto,
    data.totalAlumnos,
    data.nombreContacto,
    data.cargoContacto,
    data.emailContacto,
    data.celularContacto,
    data.ejecutivoUDD.nombre,
    data.ejecutivoUDD.email,
    data.ejecutivoUDD.celular,
    new Date()
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"result":"success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
